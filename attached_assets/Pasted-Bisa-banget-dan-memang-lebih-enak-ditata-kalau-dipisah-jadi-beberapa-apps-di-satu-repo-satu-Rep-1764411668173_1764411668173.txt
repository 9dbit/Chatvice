Bisa banget, dan memang lebih enak ditata kalau dipisah jadi beberapa “apps” di satu repo / satu Replit:
	•	/apps/landing → website Jeany AI
	•	/apps/merchant-panel → dashboard merchant
	•	/apps/superadmin-panel → dashboard super admin
	•	/apps/widget → chat widget Jeany yang ditanam di website merchant
	•	/api → backend API (chat, triggers, knowledge, supervisor, dll.)
	•	/services & /database → logic & penyimpanan data (Replit DB / Supabase)

Ini tetap 1 project Replit, 1 server NodeJS, tapi struktur file rapi dan terpisah per aplikasi.
Sangat memungkinkan dan efektif untuk:
	•	memudahkan tracing file
	•	bisa scale belakangan (kalau mau pecah jadi microservice beneran pun gampang)
	•	developer lain lebih gampang onboarding.

🧠 5. AI ENGINE + SESSION + TRIGGER

services/ai-engine.js

import OpenAI from "openai";
import { checkTriggers } from "./trigger.js";
import { setHumanMode, getSession } from "./session.js";

const client = new OpenAI({ apiKey: process.env.OPENAI_KEY });

export async function askJeany({ session_id, merchant_id, message, knowledge }) {
  const session = await getSession(session_id, merchant_id);

  // kalau sudah HUMAN mode → jangan panggil AI
  if (session.mode === "HUMAN") {
    return {
      answer: "👩‍💼 Supervisor sedang menangani percakapan ini.",
      mode: "HUMAN"
    };
  }

  // cek trigger kata
  const isTriggered = await checkTriggers(merchant_id, message);
  if (isTriggered) {
    await setHumanMode(session_id, merchant_id);
    return {
      answer: "⚠️ Saya akan menghubungkan Anda dengan Supervisor agar bisa dibantu lebih lanjut.",
      mode: "HUMAN"
    };
  }

  // prompt AI
  const prompt = `
You are Jeany, an AI Customer Service Agent.
You are friendly, talkative, and helpful.
Always answer in a clear, structured way, but human-like.

Merchant knowledge:
${knowledge}

Customer: ${message}
Jeany:
`;

  const completion = await client.chat.completions.create({
    model: "gpt-4.1-mini",
    messages: [{ role: "user", content: prompt }],
    temperature: 0.7
  });

  const answer = completion.choices[0].message.content;
  return { answer, mode: "AI" };
}

services/session.js

import db from "../database/db.js";
import { notifySupervisors } from "./trigger.js";

export async function getSession(session_id, merchant_id) {
  let session = await db.get(`session_${session_id}`);
  if (!session) {
    session = {
      id: session_id,
      merchant_id,
      mode: "AI",
      supervisor_id: null,
      created_at: Date.now(),
      updated_at: Date.now()
    };
    await db.set(`session_${session_id}`, session);
  }
  return session;
}

export async function setHumanMode(session_id, merchant_id) {
  const session = await getSession(session_id, merchant_id);
  session.mode = "HUMAN";
  session.updated_at = Date.now();
  await db.set(`session_${session_id}`, session);
  await notifySupervisors(merchant_id, session_id);
}

services/trigger.js

import db from "../database/db.js";

export async function checkTriggers(merchant_id, text) {
  const list = (await db.get(`triggers_${merchant_id}`)) || [
    "depo belum masuk",
    "withdraw belum masuk",
    "ganti password"
  ];
  const lower = text.toLowerCase();
  return list.some(t => lower.includes(t.toLowerCase()));
}

export async function notifySupervisors(merchant_id, session_id) {
  const supervisors = (await db.get(`supervisors_${merchant_id}`)) || [];
  for (const sup of supervisors) {
    await db.push(`notif_${sup.id}`, {
      session_id,
      message: "Customer membutuhkan superviser (trigger detected)",
      seen: false,
      created_at: Date.now()
    });
  }
}

🧵 6. API: CHAT – api/chat.js

import express from "express";
import db from "../database/db.js";
import { askJeany } from "../services/ai-engine.js";

const router = express.Router();

router.post("/ask", async (req, res) => {
  try {
    const { merchant_id, session_id, message } = req.body;
    if (!merchant_id || !session_id || !message) {
      return res.status(400).json({ error: "missing fields" });
    }

    const knowledge = (await db.get(`knowledge_${merchant_id}`)) || "";
    const result = await askJeany({ session_id, merchant_id, message, knowledge });

    // log chat
    const logKey = `chatlog_${session_id}`;
    const log = (await db.get(logKey)) || [];
    log.push({
      from: "user",
      message,
      time: Date.now()
    });
    log.push({
      from: result.mode === "HUMAN" ? "system" : "jeany",
      message: result.answer,
      time: Date.now()
    });
    await db.set(logKey, log);

    res.json({
      answer: result.answer,
      mode: result.mode
    });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: "server error" });
  }
});

export default router;

api/merchant.js (settings merchant + icon + theme)

import express from "express";
import db from "../database/db.js";

const router = express.Router();

// get merchant config (dipakai widget)
router.get("/status/:merchant_id", async (req, res) => {
  const merchant_id = req.params.merchant_id;
  const config = (await db.get(`merchant_${merchant_id}`)) || {
    icon_url: "https://via.placeholder.com/80x80.png?text=CS",
    icon_size: 70,
    online: true,
    name: "Default Merchant"
  };
  res.json(config);
});

// update config
router.post("/config", async (req, res) => {
  const { merchant_id, icon_url, icon_size, online } = req.body;
  const current = (await db.get(`merchant_${merchant_id}`)) || {};
  const updated = {
    ...current,
    icon_url: icon_url || current.icon_url,
    icon_size: icon_size || current.icon_size || 70,
    online: typeof online === "boolean" ? online : (current.online ?? true)
  };
  await db.set(`merchant_${merchant_id}`, updated);
  res.json({ success: true, config: updated });
});

export default router;

const merchant_id = prompt("Masukkan Merchant ID:");

document.getElementById("merchantIdLabel").innerText = merchant_id;

// Save knowledge
document.getElementById("saveKnowledge").onclick = async () => {
  const text = document.getElementById("knowledgeText").value;
  await fetch("/api/knowledge/set", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ merchant_id, knowledge_text: text })
  });
  alert("Knowledge disimpan");
};

// Triggers (simple: simpan array)
let triggers = [];

document.getElementById("addTrigger").onclick = async () => {
  const t = document.getElementById("triggerInput").value.trim();
  if (!t) return;
  triggers.push(t);
  renderTriggers();
  await fetch("/api/knowledge/set", { method: "POST" }); // TODO: endpoint khusus triggers
};

function renderTriggers() {
  const ul = document.getElementById("triggerList");
  ul.innerHTML = "";
  triggers.forEach((t, i) => {
    const li = document.createElement("li");
    li.innerText = t;
    ul.appendChild(li);
  });
}

// Branding save
document.getElementById("saveBranding").onclick = async () => {
  const icon_url = document.getElementById("iconUrl").value;
  const icon_size = parseInt(document.getElementById("iconSize").value) || 70;
  const online = document.getElementById("onlineStatus").checked;

    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ merchant_id, icon_url, icon_size, online })
  });
  const data = await res.json();
  alert("Branding disimpan");
